def _migrate_columns(conn):
    """পুরোনো DB-তে নতুন কলাম যোগ করে (চ্যানেল গোডাউন: chan_file_id/chan_msg_id)।"""
    cols = {r[1] for r in conn.execute("PRAGMA table_info(tgfiles)")}
    if "chan_file_id" not in cols:
        conn.execute("ALTER TABLE tgfiles ADD COLUMN chan_file_id TEXT")
    if "chan_msg_id" not in cols:
        conn.execute("ALTER TABLE tgfiles ADD COLUMN chan_msg_id INTEGER")
    conn.commit()


def _db():
    conn = sqlite3.connect(DB_PATH)
    conn.execute(
        "CREATE TABLE IF NOT EXISTS users ("
        "id INTEGER PRIMARY KEY, first_seen TEXT, name TEXT)"
    )
    conn.execute(
        "CREATE TABLE IF NOT EXISTS tgfiles ("
        "code TEXT PRIMARY KEY, owner INTEGER, file_id TEXT, name TEXT, "
        "size INTEGER, kind TEXT, t TEXT)"
    )
    conn.execute(
        "CREATE TABLE IF NOT EXISTS pending ("
        "user_id INTEGER PRIMARY KEY, file_id TEXT, name TEXT, "
        "size INTEGER, kind TEXT, ts INTEGER)"
    )
    conn.execute(
        "CREATE TABLE IF NOT EXISTS meta (key TEXT PRIMARY KEY, value TEXT)"
    )
    _migrate_columns(conn)
    return conn


def track_user(user) -> bool:
    """True if this user is brand new."""
    try:
        conn = _db()
        cur = conn.execute(
            "INSERT OR IGNORE INTO users (id, first_seen, name) VALUES (?, datetime('now'), ?)",
            (user.id, user.first_name or ""),
        )
        conn.commit()
        is_new = cur.rowcount > 0
        conn.close()
        return is_new
    except Exception:
        log.exception("track_user failed")
        return False


def user_count() -> int:
    try:
        conn = _db()
        n = conn.execute("SELECT COUNT(*) FROM users").fetchone()[0]
        conn.close()
        return n
    except Exception:
        return -1

def human_size(n) -> str:
    try:
        n = float(n or 0)
    except Exception:
        return "?"
    if n < 1024:
        return f"{int(n)} B"
    if n < 1024 * 1024:
        return f"{n / 1024:.1f} KB"
    return f"{n / (1024 * 1024):.2f} MB"


def new_code(n: int = 8) -> str:
    return "".join(secrets.choice(CODE_ALPHABET) for _ in range(n))


def set_pending(user_id: int, file_id: str, name: str, size, kind: str):
    conn = _db()
    conn.execute(
        "INSERT OR REPLACE INTO pending (user_id, file_id, name, size, kind, ts) "
        "VALUES (?, ?, ?, ?, ?, ?)",
        (user_id, file_id, name or "", int(size or 0), kind, int(time.time())),
    )
    conn.commit()
    conn.close()


def get_pending(user_id: int):
    conn = _db()
    row = conn.execute(
        "SELECT file_id, name, size, kind, ts FROM pending WHERE user_id = ?",
        (user_id,),
    ).fetchone()
    conn.close()
    if not row:
        return None
    file_id, name, size, kind, ts = row
    if int(time.time()) - int(ts or 0) > PENDING_TTL:
        clear_pending(user_id)
        return None
    return {"file_id": file_id, "name": name, "size": size, "kind": kind}


def clear_pending(user_id: int):
    conn = _db()
    conn.execute("DELETE FROM pending WHERE user_id = ?", (user_id,))
    conn.commit()
    conn.close()


def store_file(user_id: int, file_id: str, name: str, size, kind: str):
    """নতুন হোস্টিং রেকর্ড বসিয়ে শেয়ার কোড ফেরত দেয়; ব্যর্থ হলে None।"""
    for _ in range(6):
        code = new_code()
        conn = None
        try:
            conn = _db()
            conn.execute(
                "INSERT INTO tgfiles (code, owner, file_id, name, size, kind, t) "
                "VALUES (?, ?, ?, ?, ?, ?, datetime('now'))",
                (code, user_id, file_id, name or "", int(size or 0), kind),
            )
            conn.commit()
            conn.close()
            return code
        except sqlite3.IntegrityError:
            try:
                if conn:
                    conn.close()
            except Exception:
                pass
            continue
        except Exception:
            log.exception("store_file failed")
            return None
    return None

def storage_channel_from_forward(msg):
    """ফরওয়ার্ড করা চ্যানেল পোস্ট থেকে চ্যানেল আইডি বের করে; না মিললে None।"""
    if msg is None:
        return None
    chat = None
    origin = getattr(msg, "forward_origin", None)
    if origin is not None:
        chat = getattr(origin, "chat", None)
    if chat is None:
        chat = getattr(msg, "forward_from_chat", None)
    if chat is None:
        return None
    ctype = str(getattr(chat, "type", "") or "")
    if "channel" not in ctype:
        return None
    cid = getattr(chat, "id", None)
    try:
        return int(cid) if cid is not None else None
    except Exception:
        return None

async def _copy_to_storage(bot, user, name: str, size, kind: str, file_id: str, code: str):
    """ফাইলটা স্টোরেজ চ্যানেলে কপি করে; (chan_file_id, chan_msg_id) ফেরত দেয়।

    চ্যানেল সেট না থাকলে বা কপি ব্যর্থ হলে (None, None) — হোস্টিং তাতে আটকায় না।
    """
    chan = get_storage_channel()
    if not chan or bot is None:
        return None, None
    caption = (
        f"📁 {name or 'file'} ({human_size(size)})\n"
        f"👤 {user.first_name or ''} (ID {user.id})\n"
        f"🔑 কোড: {code}"
    )
    try:
        if kind == "photo":
            sent = await bot.send_photo(chan, file_id, caption=caption)
        elif kind == "video":
            sent = await bot.send_video(chan, file_id, caption=caption)
        elif kind == "audio":
            sent = await bot.send_audio(chan, file_id, caption=caption)
        elif kind == "voice":
            sent = await bot.send_voice(chan, file_id)
        else:
            sent = await bot.send_document(chan, file_id, caption=caption)
        return _sent_file_id(sent, kind), getattr(sent, "message_id", None)
    except Exception:
        log.exception("storage channel copy failed")
        return None, None


# ---------- DB ব্যাকআপ / রিস্টোর (হোস্ট বদল সুরক্ষা) ----------

def _db_snapshot_bytes():
    """users.db-র consistent snapshot বানিয়ে bytes ফেরত দেয়; ব্যর্থ হলে None।"""
    tmp = DB_PATH + ".snap"
    try:
        if os.path.exists(tmp):
            os.remove(tmp)
        src = sqlite3.connect(DB_PATH)
        dst = sqlite3.connect(tmp)
        try:
            src.backup(dst)
        finally:
            dst.close()
            src.close()
        with open(tmp, "rb") as fh:
            data = fh.read()
        os.remove(tmp)
        return data
    except Exception:
        log.exception("db snapshot failed")
        try:
            if os.path.exists(tmp):
                os.remove(tmp)
        except Exception:
            pass
        return None


def _validate_backup_bytes(data: bytes):
    """ব্যাকআপ bytes ঠিকমতো SQLite কিনা + দরকারি টেবিল আছে কিনা; ঠিক হলে counts, নাহলে None।"""
    if not data or not data.startswith(b"SQLite format 3\x00"):
        return None
    tmp = None
    try:
        fd, tmp = tempfile.mkstemp(suffix=".db")
        with os.fdopen(fd, "wb") as fh:
            fh.write(data)
        conn = sqlite3.connect(tmp)
        try:
            tables = {
                r[0]
                for r in conn.execute(
                    "SELECT name FROM sqlite_master WHERE type='table'"
                )
            }
            if not {"tgfiles", "users", "pending"}.issubset(tables):
                return None
            files = conn.execute("SELECT COUNT(*) FROM tgfiles").fetchone()[0]
            users = conn.execute("SELECT COUNT(*) FROM users").fetchone()[0]
        finally:
            conn.close()
        return {"files": files, "users": users}
    except Exception:
        return None
    finally:
        if tmp and os.path.exists(tmp):
            try:
                os.remove(tmp)
            except Exception:
                pass


async def send_db_backup(bot, reason: str) -> bool:

    return sent_any

async def on_admin_restore(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """এডমিন ব্যাকআপ .db ফাইল পাঠালে DB রিস্টোর হয় — হোস্ট বদলের পর পুরোনো লিংক বাঁচাতে।"""
    msg = update.message
    try:
        tg_file = await context.bot.get_file(msg.document.file_id)
        data = bytes(await tg_file.download_as_bytearray())
    except Exception:
        log.exception("restore: download failed")
        await msg.reply_text("❌ ব্যাকআপ ফাইলটা নামানো যাচ্ছে না — আবার পাঠাও।")
        return
    info = _validate_backup_bytes(data)
    if not info:
        await msg.reply_text(
            "❌ এটা ঠিক ব্যাকআপ ফাইল মনে হচ্ছে না — রিস্টোর হয়নি, আগের ডাটাই আছে।"
        )
        return
    try:
        if os.path.exists(DB_PATH):
            shutil.copyfile(DB_PATH, DB_PATH + ".pre-restore.bak")
        tmp_path = DB_PATH + ".restore-tmp"
        with open(tmp_path, "wb") as fh:
            fh.write(data)
        os.replace(tmp_path, DB_PATH)
    except Exception:
        log.exception("restore: write failed")
        await msg.reply_text("❌ রিস্টোর লিখতে সমস্যা হয়েছে — আবার চেষ্টা করো।")
        return
    log.info("DB restored by admin: %s files, %s users", info["files"], info["users"])
    await msg.reply_text(
        "✅ <b>ব্যাকআপ রিস্টোর হয়ে গেছে!</b>\n"
        f"📁 হোস্ট করা ফাইল: {info['files']} · 👥 ইউজার: {info['users']}\n"
        "আগের সব শেয়ার লিংক আবার কাজ করবে।",
        parse_mode=ParseMode.HTML,
    )

def _fb_request(method: str, path: str, payload=None, timeout: int = 8):
    url = FB_DB + path
    data = None
    headers = {}
    if payload is not None:
        data = json.dumps(payload).encode("utf-8")
        headers["Content-Type"] = "application/json"
    req = urllib.request.Request(url, data=data, headers=headers, method=method)
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        body = resp.read().decode("utf-8")
        return json.loads(body) if body else None


def fb_get_key(code: str):
    try:
        return _fb_request("GET", "/keys/" + urllib.parse.quote(code, safe="") + ".json")
    except Exception:
        return None


def fb_burn_key(code: str, user_id: int) -> bool:
    try:
        _fb_request(
            "PATCH",
            "/keys/" + urllib.parse.quote(code, safe="") + ".json",
            {
                "used": True,
                "usedFor": "file-host-tg",
                "usedAt": int(time.time() * 1000),
                "usedBy": user_id,
            },
        )
        return True
    except Exception:
        return False


def try_consume_key(code: str, user_id: int) -> str:
    """'ok' | 'notfound' | 'used' | 'error'"""
    rec = fb_get_key(code)
    if not rec:
        return "notfound"
    if rec.get("used") is True:
        return "used"
    if not fb_burn_key(code, user_id):
        return "error"
    return "ok"

async def is_member(bot, user_id: int):
    """True/False; None if the check itself failed (e.g. bot not channel admin)."""
    try:
        member = await bot.get_chat_member(CHANNEL, user_id)
    except Exception as exc:
        log.warning("getChatMember failed for %s: %s", user_id, exc)
        return None
    if member.status in (
        ChatMemberStatus.OWNER,
        ChatMemberStatus.ADMINISTRATOR,
        ChatMemberStatus.MEMBER,
    ):
        return True
    if member.status == ChatMemberStatus.RESTRICTED:
        return bool(getattr(member, "is_member", False))
    return False

            elif rec["kind"] == "video":
                await update.message.reply_video(fid, caption=caption, parse_mode=ParseMode.HTML)
            elif rec["kind"] == "audio":
                await update.message.reply_audio(fid, caption=caption, parse_mode=ParseMode.HTML)
            elif rec["kind"] == "voice":
                await update.message.reply_voice(fid)
            else:
                await update.message.reply_document(fid, caption=caption, parse_mode=ParseMode.HTML)
        except Exception:
            chan_fid = rec.get("chan_file_id")

            if chan_fid and chan_fid != fid:

                    if rec["kind"] == "photo":
                        await update.message.reply_photo(chan_fid, caption=caption, parse_mode=ParseMode.HTML)

                await update.message.reply_text(

def _media_info(msg):
    """message থেকে (file_id, name, size, kind); কিছু না মিললে None।"""
    if msg.document:
        d = msg.document
        return d.file_id, (d.file_name or "file"), d.file_size, "document"
    if msg.video:
        v = msg.video
        return v.file_id, (v.file_name or "video.mp4"), v.file_size, "video"
    if msg.audio:
        a = msg.audio
        return a.file_id, (a.file_name or (a.title or "audio")), a.file_size, "audio"
    if msg.voice:
        return msg.voice.file_id, "voice.ogg", msg.voice.file_size, "voice"
    if getattr(msg, "animation", None):
        an = msg.animation
        return an.file_id, (an.file_name or "animation.mp4"), an.file_size, "document"
    if msg.photo:
        p = msg.photo[-1]
        return p.file_id, "photo.jpg", p.file_size, "photo"
    return None

async def on_text_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user
    msg = update.message
    if user is None or msg is None or not msg.text:
        return
    text = msg.text.strip()
    if user.id in ADMIN_IDS:
        chan_id = storage_channel_from_forward(msg)
        if chan_id is not None:
            set_storage_channel(chan_id)
            log.info("storage channel set to %s via forward", chan_id)
            await msg.reply_text(

            )
            return

    if pend:
        if len(text) < 4 or len(text) > 64 or " " in text:
            await msg.reply_text("🔑 শুধু কিটাই পাঠাও — বাড়তি লেখা ছাড়া।")

            )

            return
        chan_fid, chan_mid = await _copy_to_storage(
            context.bot, user, pend["name"], pend["size"], pend["kind"], pend["file_id"], code
        )
        if chan_fid or chan_mid:
            set_file_channel(code, chan_fid, chan_mid)

            f"✅ <b>হোস্ট হয়ে গেছে!</b> ({human_size(pend['size'])})\n\n"
            f"🔗 শেয়ার লিংক:\n{share_link(code)}\n\n"
            "লিংকটা যাকে দেবে সেও ফাইলটা পেয়ে যাবে (শুধু @rashdteem মেম্বার হতে হবে)। মেয়াদ নেই — ফাইল Telegram-এই থাকবে।"
            f"{godown_line}\n"
            f"তোমার ফাইল: {user_file_count(user.id)}/{MAX_USER_FILES} — সব দেখো: /myfiles",
            parse_mode=ParseMode.HTML,
        )
        try:
            await send_db_backup(context.bot, "নতুন ফাইল হোস্ট হয়েছে")
        except Exception:
            log.exception("auto backup after store failed")
        return
    if 6 <= len(text) <= 20 and all(ch.isalnum() or ch in "-_" for ch in text):
        await msg.reply_text(
            "📤 আগে আমাকে একটা ফাইল পাঠাও (2GB-এর নিচে) — তারপর কি দিলেই হোস্ট করে লিংক দিয়ে দেব।"
        )


async def cmd_myfiles(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user
    if user is None or update.message is None:
        return
    files = list_user_files(user.id, limit=MAX_USER_FILES)
    count = user_file_count(user.id)
    if not files:
        await update.message.reply_text(
            f"📭 এখনো কোনো ফাইল হোস্ট করোনি। (তোমার লিমিট: সর্বোচ্চ {MAX_USER_FILES}টা ফাইল)\n\n"
            "যেকোনো ফাইল (2GB-এর নিচে) এই চ্যাটে পাঠালেই হোস্ট করা শুরু হবে।"
        )
        return
    lines = [f"📁 <b>তোমার হোস্ট করা ফাইলগুলো ({count}/{MAX_USER_FILES}):</b>\n"]
    for i, f in enumerate(files, 1):
        nm = (f["name"] or "file").replace("<", "").replace(">", "")
        lines.append(f"{i}. <b>{nm}</b> ({human_size(f['size'])})\n{share_link(f['code'])}")
    rows = [
        [InlineKeyboardButton(f"🗑 {i}", callback_data=f"del_{f['code']}")]
        for i, f in enumerate(files, 1)
    ]
    await update.message.reply_text(
        "\n".join(lines) + "\n\nডিলিট করতে নিচে ফাইলের নম্বরের বাটন চাপো 👇",
        parse_mode=ParseMode.HTML,
        reply_markup=InlineKeyboardMarkup(rows),
    )


async def cb_delfile(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    if query is None or not query.data:
        return
    code = query.data[4:]
    user_id = query.from_user.id
    chan = get_storage_channel()
    rec = delete_user_file(user_id, code)
    if not rec:
        await query.answer("এই ফাইলটা তোমার না, বা আগেই ডিলিট হয়ে গেছে।", show_alert=True)
        return
    if chan and rec.get("chan_msg_id"):
        try:
            await context.bot.delete_message(chan, rec["chan_msg_id"])
        except Exception:
            log.warning("channel copy delete failed for %s", code)
    await query.answer("🗑 ফাইল ডিলিট হয়েছে")
    try:
        safe_name = (rec["name"] or "file").replace("<", "").replace(">", "")
        await context.bot.send_message(
            user_id,
            f"🗑 <b>{safe_name}</b> ডিলিট হয়ে গেছে।\n"
            f"এখন তোমার ফাইল: {user_file_count(user_id)}/{MAX_USER_FILES}",
            parse_mode=ParseMode.HTML,
        )
    except Exception:
        pass


# ---------- startup: bio / description / photo / commands ----------

async def on_startup(app: Application):
    bot = app.bot
    try:
        me = await bot.get_me()
        log.info("[TOKEN CHECK] SUCCESS. Bot Username: @%s", me.username)
    except Exception:
        log.exception("get_me failed — token check failed")
    try:
        await bot.set_my_commands(
            [
                BotCommand("start", "🚀 বট চালু করুন"),
                BotCommand("myfiles", "📁 আমার হোস্ট করা ফাইল"),
                BotCommand("help", "📖 সাহায্য"),
                BotCommand("backup", "🗄️ DB ব্যাকআপ (এডমিন)"),
            ]
        )
    except Exception:
        log.exception("set_my_commands failed")
    try:
        await bot.set_my_short_description(BOT_SHORT_DESCRIPTION)
    except Exception:
        log.exception("set_my_short_description failed")
    try:
        await bot.set_my_description(BOT_DESCRIPTION)
    except Exception:
        log.exception("set_my_description failed")
    try:
        if os.path.exists(LOGO_PATH):
            with open(LOGO_PATH, "rb") as fh:
                await bot.set_my_profile_photo(
                    photo=InputProfilePhotoStatic(photo=fh.read())
                )
            log.info("Profile photo updated from logo.png")
    except Exception:
        log.exception("set_my_profile_photo failed (BotFather থেকে ম্যানুয়ালি দেওয়া যাবে)")


def main():
    if not BOT_TOKEN:
        raise SystemExit("BOT_TOKEN env var নেই — panel ENV-তে বটের টোকেন বসাতে হবে")
    app = (
        Application.builder()
        .token(BOT_TOKEN)
        .post_init(on_startup)
        .build()
    )
    app.add_handler(CommandHandler("start", cmd_start))
    app.add_handler(CommandHandler("help", cmd_help))
    app.add_handler(CommandHandler("myfiles", cmd_myfiles))
    app.add_handler(CommandHandler("admin", cmd_admin))
    app.add_handler(CommandHandler("backup", cmd_backup))
    app.add_handler(CallbackQueryHandler(cb_check_join, pattern="^check_join$"))
    app.add_handler(CallbackQueryHandler(cb_delfile, pattern="^del_"))
    app.add_handler(MessageHandler(filters.ATTACHMENT, on_media_router))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, on_text_message))
    log.info("RASHD Hosting Bot started (@%s)", BOT_USERNAME)
    app.run_polling(allowed_updates=Update.ALL_TYPES, drop_pending_updates=True)


if __name__ == "__main__":
    main()
